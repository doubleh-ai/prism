**Goal:** Add a pluggable market data layer that exposes real-time quotes, K-line, fundamentals, and ticker search through 5 new FastAPI endpoints, with a Protocol-driven provider chain, SQLite TTL cache, and graceful failover.

**Architecture:** `MarketDataProvider` Protocol defines the contract; `ProviderChain` tries providers in priority order and falls back to stale cache on total failure. All external data flows through `MarketCache` (TTL-keyed SQLite). Frontend embeds `[[quote:TICKER]]` tags in chat output; the renderer queries `/api/quote/{ticker}` and injects a price chip inline. Provider libraries (`akshare`, `baostock`) are **optional dependencies** — missing imports skip the provider silently; cache keeps the system responsive.

**Tech Stack:** FastAPI (existing), Pydantic v2 (existing), SQLite WAL (existing), `akshare>=1.13,<2` and `baostock>=0.8.11,<1` (optional, in `requirements-extra.txt`), vanilla JS (existing).

```python
"""Pydantic schemas for market data.

All schemas are strictly typed so the router can serialize directly
without ad-hoc dict manipulation. Optional fields default to ``None``
to accommodate partial responses from real-world data providers.
"""
from typing import Optional
from pydantic import BaseModel, Field


class Quote(BaseModel):
    """Real-time quote for a single ticker."""
    ticker: str = Field(..., min_length=1, max_length=32)
    name: Optional[str] = Field(default=None, max_length=128)
    price: float = Field(..., ge=0)
    change: float = 0.0  # absolute change vs prev close
    change_pct: float = 0.0  # percentage change
    volume: float = Field(default=0.0, ge=0)
    turnover: float = Field(default=0.0, ge=0)
    high: Optional[float] = Field(default=None, ge=0)
    low: Optional[float] = Field(default=None, ge=0)
    open: Optional[float] = Field(default=None, ge=0)
    prev_close: Optional[float] = Field(default=None, ge=0)
    timestamp: int = Field(..., gt=0)  # unix seconds
    source: str = Field(..., min_length=1, max_length=32)
    stale: bool = False  # True when served from expired cache after provider failure


class KLine(BaseModel):
    """Single OHLCV bar."""
    timestamp: int = Field(..., gt=0)
    open: float
    high: float
    low: float
    close: float
    volume: float = Field(..., ge=0)
    turnover: Optional[float] = Field(default=None, ge=0)


class Fundamentals(BaseModel):
    """Fundamental indicators for a single ticker."""
    ticker: str = Field(..., min_length=1, max_length=32)
    name: Optional[str] = Field(default=None, max_length=128)
    pe_ttm: Optional[float] = None
    pe_static: Optional[float] = None
    pb: Optional[float] = None
    ps_ttm: Optional[float] = None
    roe: Optional[float] = None
    revenue: Optional[float] = None
    net_profit: Optional[float] = None
    market_cap: Optional[float] = None
    timestamp: int = Field(..., gt=0)
    source: str = Field(..., min_length=1, max_length=32)


class SearchHit(BaseModel):
    """Single ticker search result."""
    ticker: str = Field(..., min_length=1, max_length=32)
    name: str = Field(..., min_length=1, max_length=128)
    market: Optional[str] = Field(default=None, max_length=8)  # e.g. "SH", "SZ", "HK"
```

```python
"""SQLite-backed TTL cache for market data.

Stored as JSON. ``get_or_fetch`` semantics are NOT here — providers
call ``get`` / ``set`` and the chain decides when to fall back to
stale. This keeps the cache free of provider-specific logic.
"""
import json
import time
from typing import Any, Optional
from ..database import get_db


class MarketCache:
    """TTL cache keyed by string. Values are JSON-serializable."""

    def get(self, key: str) -> tuple[Optional[Any], bool]:
        """Return (value, stale).

        ``stale`` is always False here — use :meth:`get_stale` for the
        fallback path that bypasses TTL.

        Note: we compare with ``time.time()`` (float seconds), not
        ``int(time.time())``. Sub-second TTLs in tests need this — a
        ttl=1 row written at t=0.3 has expires_at=1 stored as int,
        and ``int(time.time()) > 1`` at t=1.2 would still evaluate
        False (1 > 1), incorrectly keeping the row alive.
        """
        with get_db() as db:
            row = db.execute(
                "SELECT value_json, expires_at FROM market_cache WHERE cache_key=?",
                [key],
            ).fetchone()
        if not row:
            return None, False
        if time.time() > row["expires_at"]:
            return None, False
        return json.loads(row["value_json"]), False

    def get_stale(self, key: str) -> tuple[Optional[Any], bool]:
        """Return (value, True) even if expired, for fallback after provider failure."""
        with get_db() as db:
            row = db.execute(
                "SELECT value_json FROM market_cache WHERE cache_key=?",
                [key],
            ).fetchone()
        if not row:
            return None, True
        return json.loads(row["value_json"]), True

    def set(self, key: str, value: Any, ttl: int) -> None:
        """Store ``value`` under ``key`` for ``ttl`` seconds."""
        now = int(time.time())
        with get_db() as db:
            db.execute(
                """
                INSERT INTO market_cache (cache_key, value_json, fetched_at, expires_at)
                VALUES (?, ?, ?, ?)
                ON CONFLICT(cache_key) DO UPDATE SET
                    value_json=excluded.value_json,
                    fetched_at=excluded.fetched_at,
                    expires_at=excluded.expires_at
                """,
                [key, json.dumps(value, ensure_ascii=False), now, now + ttl],
            )

    def invalidate(self, key: str) -> int:
        """Delete a single key. Returns rows affected (0 or 1)."""
        with get_db() as db:
            cur = db.execute("DELETE FROM market_cache WHERE cache_key=?", [key])
        return cur.rowcount
```

```python
"""Provider Protocol — the contract every data source must satisfy.

Provider implementations live in sibling modules (``akshare_provider``,
``baostock_provider``). They may be sync or async internally; the
public surface here is async, and the chain handles mixing.
"""
from typing import Protocol, runtime_checkable, List
from .models import Quote, KLine, Fundamentals, SearchHit


class ProviderError(Exception):
    """Base class for all provider errors."""


class ProviderNotFound(ProviderError):
    """The provider has no data for the requested ticker (404 equivalent)."""


class ProviderUnavailable(ProviderError):
    """The provider is up but failed (timeout, network, rate limit, ImportError)."""


class NoProviderAvailable(ProviderError):
    """All providers in the chain failed and no cache fallback exists."""


@runtime_checkable
class MarketDataProvider(Protocol):
    """The contract every market data provider must satisfy."""
    name: str

    async def get_quote(self, ticker: str) -> Quote: ...
    async def get_kline(
        self, ticker: str, period: str = "1d", limit: int = 120
    ) -> List[KLine]: ...
    async def get_fundamentals(self, ticker: str) -> Fundamentals: ...
    async def search(self, query: str, limit: int = 10) -> List[SearchHit]: ...
    async def health(self) -> bool: ...
```

```python
"""ProviderChain — failover order, dead-provider skip, stale fallback."""
import pytest
import asyncio
from server.market.chain import ProviderChain
from server.market.cache import MarketCache
from server.market.provider import (
    MarketDataProvider, ProviderUnavailable, NoProviderAvailable,
)
from server.market.models import Quote, KLine, Fundamentals, SearchHit


class FakeProvider:
    def __init__(self, name, quote=None, fail=False, raise_notfound=False):
        self.name = name
        self._quote = quote or Quote(
            ticker="X", price=1.0, timestamp=1, source=name,
        )
        self._fail = fail
        self._raise_notfound = raise_notfound
        self.calls = 0

    async def get_quote(self, ticker):
        self.calls += 1
        if self._raise_notfound:
            raise ProviderUnavailable(f"{self.name} simulated not-found")
        if self._fail:
            raise ProviderUnavailable(f"{self.name} down")
        return self._quote.model_copy(update={"source": self.name, "ticker": ticker})

    async def get_kline(self, ticker, period="1d", limit=120):
        if self._fail:
            raise ProviderUnavailable(f"{self.name} down")
        return []

    async def get_fundamentals(self, ticker):
        if self._fail:
            raise ProviderUnavailable(f"{self.name} down")
        return Fundamentals(ticker=ticker, timestamp=1, source=self.name)

    async def search(self, query, limit=10):
        if self._fail:
            raise ProviderUnavailable(f"{self.name} down")
        return []

    async def health(self):
        return not self._fail


@pytest.mark.asyncio
async def test_first_provider_succeeds():
    cache = MarketCache()
    a = FakeProvider("a")
    b = FakeProvider("b")
    chain = ProviderChain([a, b], cache=cache)
    quote, stale = await chain.get_quote("600519")
    assert quote.source == "a"
    assert stale is False
    assert a.calls == 1
    assert b.calls == 0
```

```python
"""ProviderChain — try providers in order, fall back to stale cache.

On a successful provider call the result is written to :class:`MarketCache`
under a stable key. On a provider failure, the chain moves to the next
provider. If every provider fails, the chain returns the most recent
cached value (even if expired) and flags ``stale=True``. If no cache
exists either, raises :class:`NoProviderAvailable`.
"""
import logging
import json
import time
from typing import List, Dict, Tuple, Any
from .provider import (
    MarketDataProvider, ProviderUnavailable, NoProviderAvailable,
)
from .cache import MarketCache
from .models import Quote, KLine, Fundamentals, SearchHit

log = logging.getLogger("stock-app")


class ProviderChain:
    """Wraps an ordered list of providers with cache-backed failover."""

    def __init__(
        self,
        providers: List[MarketDataProvider],
        cache: MarketCache,
        ttls: Dict[str, int] | None = None,
    ):
        self.providers = providers
        self.cache = cache
        self.ttls = ttls or {}  # {"quote": 60, "kline": 300, "fundamentals": 86400}

    def _cache_key(self, method: str, provider_name: str, *parts: str) -> str:
        return f"{method}:{provider_name}:" + ":".join(parts)

    async def _with_failover(
        self,
        method: str,
        cache_parts: tuple,
        invoke,  # async callable: (provider) -> result
        ttl_attr: str,
    ) -> Tuple[Any, bool]:
        """Run ``invoke(p)`` over each provider; cache success; fall back to stale.

        ``invoke`` is called as ``await invoke(provider)``.
        """
        last_err: Exception | None = None
        for p in self.providers:
            try:
                result = await invoke(p)
            except ProviderUnavailable as e:
                log.warning(f"market provider {p.name} failed: {e}")
                last_err = e
                continue
            # Success — cache the JSON-shaped value
            key = self._cache_key(method, p.name, *cache_parts)
            ttl = self.ttls.get(ttl_attr, 60)
            try:
                payload = json.loads(result.model_dump_json())
            except Exception:
                payload = result
            self.cache.set(key, payload, ttl=ttl)
            # Pydantic re-validate so we return the typed model with stale=False
            if isinstance(result, Quote):
                return Quote(**{**payload, "stale": False, "source": p.name}), False
            if isinstance(result, Fundamentals):
                return Fundamentals(**{**payload, "stale": False, "source": p.name}), False
            return result, False

        # All providers failed — try stale cache (any provider that had a value)
        for p in self.providers:
            key = self._cache_key(method, p.name, *cache_parts)
            val, _ = self.cache.get_stale(key)
            if val is not None:
                if method == "quote":
                    return Quote(**{**val, "stale": True}), True
                if method == "fundamentals":
                    return Fundamentals(**{**val, "stale": True}), True
                return val, True
        raise NoProviderAvailable(
            f"all providers failed: {[p.name for p in self.providers]}"
        ) from last_err

    async def get_quote(self, ticker: str) -> Tuple[Quote, bool]:
        # Fresh cache check before trying providers
        for p in self.providers:
            key = self._cache_key("quote", p.name, ticker)
            val, _ = self.cache.get(key)
            if val is not None:
                return Quote(**{**val, "stale": False}), False
        try:
            return await self._with_failover(
                "quote", (ticker,),
                lambda p: p.get_quote(ticker), "quote",
            )
        except NoProviderAvailable:
            # All providers failed — try stale cache, marking stale=True
            for p in self.providers:
                key = self._cache_key("quote", p.name, ticker)
                val, _ = self.cache.get_stale(key)
                if val is not None:
                    return Quote(**{**val, "stale": True}), True
            raise

    async def get_kline(
        self, ticker: str, period: str = "1d", limit: int = 120
    ) -> Tuple[List[KLine], bool]:
        # Note: KLine is a list; cache the list of dicts, return as KLine list.
        async def _invoke(p):
            kls = await p.get_kline(ticker, period=period, limit=limit)
            return [KLine(**k.model_dump()) for k in kls]
        # Custom path — bypass _with_failover's list handling
        last_err = None

        # Check cache first
        for p in self.providers:
            key = self._cache_key("kline", p.name, ticker, period, str(limit))
            val, _ = self.cache.get(key)
            if val is not None:
                return [KLine(**k) for k in val], False

        for p in self.providers:
            try:
                kls = await _invoke(p)
            except ProviderUnavailable as e:
                last_err = e
                continue
            key = self._cache_key("kline", p.name, ticker, period, str(limit))
            self.cache.set(key, [k.model_dump() for k in kls], ttl=self.ttls.get("kline", 300))
            return kls, False
        for p in self.providers:
            key = self._cache_key("kline", p.name, ticker, period, str(limit))
            val, _ = self.cache.get_stale(key)
            if val is not None:
                return [KLine(**k) for k in val], True
        raise NoProviderAvailable(
            f"all providers failed: {[p.name for p in self.providers]}"
        ) from last_err

    async def get_fundamentals(self, ticker: str) -> Tuple[Fundamentals, bool]:
        return await self._with_failover(
            "fundamentals", (ticker,),
            lambda p: p.get_fundamentals(ticker), "fundamentals",
        )

    async def search(self, query: str, limit: int = 10) -> List[SearchHit]:
        """Search is best-effort across all providers; returns first non-empty list."""
        for p in self.providers:
            try:
                hits = await p.search(query, limit=limit)
                if hits:
                    return hits
            except ProviderUnavailable as e:
                log.warning(f"search provider {p.name} failed: {e}")
                continue
        return []

    async def health(self) -> Dict[str, bool]:
        return {p.name: await p.health() for p in self.providers}
```

```python
                        volume=float(row["volume"]),
                        turnover=float(row.get("amount") or 0) or None,
                    ))
                except Exception:
                    continue
            return out[-limit:]
        except Exception as e:
            raise ProviderUnavailable(f"baostock kline failed: {e}") from e
        finally:
```

```python
"""Build the default provider chain based on ``MARKET_PROVIDERS`` config."""
import logging
from .cache import MarketCache
from .chain import ProviderChain
from .provider import ProviderError
from ..config import MARKET_PROVIDERS, QUOTE_TTL, KLINE_TTL, FUNDAMENTALS_TTL

log = logging.getLogger("stock-app")

_chain: ProviderChain | None = None


def _try_build_provider(name: str):
    """Instantiate a provider by name; return None if the library is missing."""
    try:
        if name == "akshare":
            from .akshare_provider import AkshareProvider
            return AkshareProvider()
        if name == "baostock":
            from .baostock_provider import BaostockProvider
            return BaostockProvider()
        log.warning(f"Unknown market provider: {name}")
        return None
    except Exception as e:
        log.warning(f"Failed to construct provider {name}: {e}")
        return None


def get_chain() -> ProviderChain:
    """Return a process-wide singleton chain; build it lazily on first call."""
    global _chain
    if _chain is not None:
        return _chain
    cache = MarketCache()
    providers = []
    for name in MARKET_PROVIDERS:
        p = _try_build_provider(name)
        if p is not None:
            providers.append(p)
    if not providers:
        log.warning("No market providers available — endpoints will return 503")
    _chain = ProviderChain(
        providers, cache=cache,
        ttls={"quote": QUOTE_TTL, "kline": KLINE_TTL, "fundamentals": FUNDAMENTALS_TTL},
    )
    return _chain


def reset_chain_for_tests() -> None:
    """Reset the singleton — used by tests that mock providers."""
    global _chain
    _chain = None
```

```python
"""FastAPI routes for market data — /api/quote, /api/quote/kline,
/api/fundamentals, /api/search, /api/data-providers.

All endpoints require authentication (the existing middleware
enforces that for any non-``/api/login`` path). The endpoints are
GET-only so no CSRF header is required.
"""
import re
import logging
from fastapi import APIRouter, Request, HTTPException
from fastapi.responses import JSONResponse
from .factory import get_chain
from .provider import NoProviderAvailable, ProviderNotFound

log = logging.getLogger("stock-app")

router = APIRouter(prefix="/api")

# Tickers: 6 digits (A-share) or HK 5-digit / US alpha-numeric.
# Loose pattern — providers will reject truly bogus ones with NotFound.
_TICKER_RE = re.compile(r"^[A-Za-z0-9.\-]{1,16}$")


def _validate_ticker(ticker: str) -> str:
    ticker = ticker.strip()
    if not _TICKER_RE.match(ticker):
        raise HTTPException(status_code=400, detail="invalid ticker")
    return ticker


@router.get("/quote/{ticker}")
async def api_quote(ticker: str, request: Request):
    ticker = _validate_ticker(ticker)
    try:
        quote, stale = await get_chain().get_quote(ticker)
    except NoProviderAvailable as e:
        return JSONResponse(
            status_code=503,
            content={"error": "no_provider_available", "detail": str(e)},
        )
    except ProviderNotFound:
        raise HTTPException(404, "ticker not found")
    return quote.model_dump()


@router.get("/quote/{ticker}/kline")
async def api_kline(
    ticker: str, request: Request,
    period: str = "1d", limit: int = 120,
):
    ticker = _validate_ticker(ticker)
    if period not in ("1d", "1w", "1m", "5m", "15m", "30m", "60m"):
        raise HTTPException(400, "invalid period")
    if limit < 1 or limit > 500:
        raise HTTPException(400, "limit must be 1..500")
    try:
        kls, stale = await get_chain().get_kline(ticker, period=period, limit=limit)
    except NoProviderAvailable as e:
        return JSONResponse(
            status_code=503,
            content={"error": "no_provider_available", "detail": str(e)},
        )
    return {"ticker": ticker, "period": period, "kline": [k.model_dump() for k in kls], "stale": stale}


@router.get("/fundamentals/{ticker}")
async def api_fundamentals(ticker: str, request: Request):
    ticker = _validate_ticker(ticker)
    try:
        fund, stale = await get_chain().get_fundamentals(ticker)
    except NoProviderAvailable as e:
        return JSONResponse(
            status_code=503,
            content={"error": "no_provider_available", "detail": str(e)},
        )
    except ProviderNotFound:
        raise HTTPException(404, "ticker not found")
    return fund.model_dump()


@router.get("/search")
async def api_search(request: Request, q: str = "", limit: int = 10):
    q = q.strip()
    if not q or len(q) > 32:
        return {"hits": []}
    if limit < 1 or limit > 50:
        limit = 10
    hits = await get_chain().search(q, limit=limit)
    return {"hits": [h.model_dump() for h in hits]}


@router.get("/data-providers")
async def api_data_providers(request: Request):
    health = await get_chain().health()
    providers = [{"name": name, "healthy": ok} for name, ok in health.items()]
    return {"providers": providers}
```

```javascript
// ── Quote tag rendering ([[quote:600519]]) ──
const QUOTE_TAG_RE = /\[\[quote:([A-Za-z0-9.\-]{1,16})\]\]/g;

function renderQuoteTags(html) {
  // Replace [[quote:TICKER]] with a placeholder span; prices filled async.
  return html.replace(QUOTE_TAG_RE, (_, ticker) => {
    return `<span class="quote-chip" data-ticker="${ticker}">` +
           `<span class="quote-chip__code">${ticker}</span>` +
           `<span class="quote-chip__price">…</span></span>`;
  });
}

async function fillQuoteChips(root) {
  const chips = root.querySelectorAll(".quote-chip:not([data-filled])");
  if (!chips.length) return;
  for (const chip of chips) {
    chip.dataset.filled = "1";
    const ticker = chip.dataset.ticker;
    try {
      const r = await App.api(`/api/quote/${encodeURIComponent(ticker)}`);
      if (!r || r.error) {
        chip.querySelector(".quote-chip__price").textContent = "—";
        continue;
      }
      const price = (r.price || 0).toFixed(2);
      const pct = (r.change_pct || 0).toFixed(2);
      const sign = r.change >= 0 ? "+" : "";
      chip.querySelector(".quote-chip__price").innerHTML =
        `${price} <span class="quote-chip__pct">(${sign}${pct}%)</span>`;
      chip.classList.add(r.stale ? "quote-chip--stale" : "quote-chip--live");
    } catch (e) {
      chip.querySelector(".quote-chip__price").textContent = "—";
    }
  }
}
```
